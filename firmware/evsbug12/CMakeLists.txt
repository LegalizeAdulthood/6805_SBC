set(TASM_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../tasm32")
set(TASM_EXE "${TASM_DIR}/TASM.EXE")
set(TASM_6805_TABLE "${TASM_DIR}/TASM05.TAB")

set(EVSBUG12_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/evsbug12.asm")
set(EVSBUG12_ROM "${CMAKE_CURRENT_SOURCE_DIR}/evsbug12.bin")
set(EVSBUG12_OUTPUT_DIR "${CMAKE_CURRENT_BINARY_DIR}")
set(EVSBUG12_BINARY "${EVSBUG12_OUTPUT_DIR}/evsbug12.bin")
set(EVSBUG12_LISTING "${EVSBUG12_OUTPUT_DIR}/evsbug12.lst")
set(EVSBUG12_EXPORTS "${EVSBUG12_OUTPUT_DIR}/evsbug12.exp")
set(EVSBUG12_SYMBOLS "${EVSBUG12_OUTPUT_DIR}/evsbug12.sym")
set(EVSBUG12_BUILD_TASM_TABLE "${EVSBUG12_OUTPUT_DIR}/TASM05.TAB")

add_custom_command(
    OUTPUT "${EVSBUG12_BUILD_TASM_TABLE}"
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${EVSBUG12_OUTPUT_DIR}"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${TASM_6805_TABLE}" "${EVSBUG12_BUILD_TASM_TABLE}"
    DEPENDS "${TASM_6805_TABLE}"
    COMMENT "Copying TASM 6805 table for EVSBUG12 build"
    VERBATIM
)

add_custom_command(
    OUTPUT "${EVSBUG12_BINARY}"
    BYPRODUCTS
        "${EVSBUG12_LISTING}"
        "${EVSBUG12_EXPORTS}"
        "${EVSBUG12_SYMBOLS}"
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${EVSBUG12_OUTPUT_DIR}"
    COMMAND "${TASM_EXE}" -05 -x -b -c -f00 -s "${EVSBUG12_SOURCE}" "${EVSBUG12_BINARY}" "${EVSBUG12_LISTING}" "${EVSBUG12_EXPORTS}" "${EVSBUG12_SYMBOLS}"
    DEPENDS
        "${EVSBUG12_SOURCE}"
        "${EVSBUG12_BUILD_TASM_TABLE}"
        "${TASM_EXE}"
    WORKING_DIRECTORY "${EVSBUG12_OUTPUT_DIR}"
    COMMENT "Assembling evsbug12.asm to binary ROM image"
    VERBATIM
)

add_custom_target(evsbug12 ALL DEPENDS "${EVSBUG12_BINARY}")

if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
